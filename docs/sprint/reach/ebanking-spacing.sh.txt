#!/usr/bin/env bash
# R0.1.a.a.a.a.a.a.a.a.a.a.a: does the pass-then-fail pattern depend on spacing between probes?
# Block A: 10 manuk a11y-dump probes, 10 s apart. Block B: 10 probes, 120 s apart.
# Block M (red mutant, run first): 10 probes at 10 s against 127.0.0.1:9, must all be recorded failed.
# A probe is FAILED only if manuk rc!=0. Every probe is written to the table, failed or not.
# usage: ebanking-spacing.sh <window-id>
# Output: docs/sprint/reach/EBANKING-SPACING-R0.1.a.a.a.a.a.a.a.a.a.a.a.tsv (chrome_ver per row).
# Exit 0 only if the mutant block recorded 10/10 failed rows and blocks A and B each recorded 10 rows.
set -u
REPO=/home/patrickd/manuk
CHROME_VER=${MANUK_CHROME_PIN:?MANUK_CHROME_PIN not set}
OUT=$REPO/docs/sprint/reach/EBANKING-SPACING-R0.1.a.a.a.a.a.a.a.a.a.a.a.tsv
WIN=${1:?window id}
PROBES=10
HOST=ebanking.bdm-sa.com
D=$(mktemp -d /tmp/ebk-spacing-XXXX)
[ -f "$OUT" ] || printf 'window\tblock\tspacing_s\tprobe\tutc\tprobe_url\tmanuk_rc\tmanuk_bytes\tmanuk_secs\trow_failed\tchrome_ver\n' > "$OUT"

run_block() {  # $1 block id, $2 spacing seconds, $3 url
  local B=$1 SP=$2 URL=$3 FAILS=0
  for P in $(seq 1 $PROBES); do
    [ "$P" -gt 1 ] && sleep "$SP"
    touch "$REPO/.git/sprint-working"
    T1=$(date +%s)
    timeout 120 "$REPO/target/release/a11y-dump" "$URL" > "$D/$B$P.out" 2>"$D/$B$P.err"; MRC=$?
    MSECS=$(( $(date +%s)-T1 )); MBYTES=$(wc -c < "$D/$B$P.out")
    if [ "$MRC" -ne 0 ]; then FAILED=1; FAILS=$((FAILS+1)); else FAILED=0; fi
    printf '%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\n' "$WIN" "$B" "$SP" "$P" "$(date -u +%FT%TZ)" "$URL" \
      "$MRC" "$MBYTES" "$MSECS" "$FAILED" "$CHROME_VER" >> "$OUT"
    echo "block $B probe $P (spacing ${SP}s): rc=$MRC bytes=$MBYTES secs=$MSECS failed=$FAILED"
  done
  echo "block=$B spacing=${SP}s probes=$PROBES failed=$FAILS"
  echo "$FAILS" > "$D/fails.$B"
}

run_block M 10 https://127.0.0.1:9/
run_block A 10 https://$HOST/
run_block B 120 https://$HOST/
echo "dir=$D"

# Red check: the mutant block must have 10 rows recorded, all failed.
MROWS=$(awk -F'\t' -v w="$WIN" '$1==w && $2=="M"' "$OUT" | wc -l)
MFAIL=$(awk -F'\t' -v w="$WIN" '$1==w && $2=="M" && $10==1' "$OUT" | wc -l)
AROWS=$(awk -F'\t' -v w="$WIN" '$1==w && $2=="A"' "$OUT" | wc -l)
BROWS=$(awk -F'\t' -v w="$WIN" '$1==w && $2=="B"' "$OUT" | wc -l)
echo "mutant rows=$MROWS failed=$MFAIL | A rows=$AROWS | B rows=$BROWS"
[ "$MROWS" -eq "$PROBES" ] && [ "$MFAIL" -eq "$PROBES" ] && [ "$AROWS" -eq "$PROBES" ] && [ "$BROWS" -eq "$PROBES" ]
exit $?
